    <script>
        let socket = null;
        const btnConnect = document.getElementById('btnConnect');
        const logContent = document.getElementById('logContent');
        const statusDot = document.getElementById('statusDot');
        const statusText = document.getElementById('statusText');
        const protocolWarning = document.getElementById('protocolWarning');

        // Check for file protocol
        if (window.location.protocol === 'file:') {
            protocolWarning.classList.add('show');
        }

        function addLog(event, data) {
            console.log(`[${event}]`, data); // Backup to browser console
            if (logContent.children.length === 1 && logContent.children[0].style.textAlign === 'center') {
                logContent.innerHTML = '';
            }
            const entry = document.createElement('div');
            entry.className = 'log-entry';
            const time = new Date().toLocaleTimeString();
            entry.innerHTML = `
                <span class="log-time">[${time}]</span>
                <span class="log-event">${event}</span>
                <pre>${JSON.stringify(data, null, 2)}</pre>
            `;
            logContent.prepend(entry);
        }

        function updateBadges(unreadCount) {
            if (!unreadCount) return;
            document.getElementById('countTotal').innerText = unreadCount.TOTAL || 0;
            document.getElementById('countAction').innerText = unreadCount.ACTION || 0;
            document.getElementById('countIncome').innerText = unreadCount.INCOME || 0;
            document.getElementById('countOpportunity').innerText = unreadCount.OPPORTUNITY || 0;
            document.getElementById('countUpdate').innerText = unreadCount.UPDATE || 0;
        }

        btnConnect.onclick = (e) => {
            if (e) e.preventDefault();

            if (socket && socket.connected) {
                socket.disconnect();
                return;
            }

            const rawUrl = document.getElementById('url').value.trim();
            const namespace = document.getElementById('namespace').value.trim();
            const token = document.getElementById('token').value.trim();

            if (!token) {
                alert('Please paste a JWT token');
                return;
            }

            btnConnect.innerText = 'Connecting...';
            btnConnect.disabled = true;

            // Combine URL and namespace
            const host = rawUrl.endsWith('/') ? rawUrl.slice(0, -1) : rawUrl;
            const ns = namespace.startsWith('/') ? namespace : '/' + namespace;

            addLog('SYSTEM', { message: 'Attempting connection', host: host, namespace: ns });

            try {
                socket = io(host + ns, {
                    auth: { token: token },
                    // transports: ['websocket'],
                    forceNew: true,
                    reconnectionAttempts: 3
                });
            } catch (err) {
                addLog('ERROR', { message: 'Failed to initialize socket', error: err.message });
                btnConnect.innerText = 'Connect';
                btnConnect.disabled = false;
                return;
            }

            socket.on('connect', () => {
                statusDot.className = 'status-dot connected';
                statusText.innerText = `Connected (ID: ${socket.id})`;
                btnConnect.innerText = 'Disconnect';
                btnConnect.className = 'disconnect';
                btnConnect.disabled = false;
                addLog('SYSTEM', { status: 'Connected successfully' });
            });

            socket.on('disconnect', (reason) => {
                statusDot.className = 'status-dot';
                statusText.innerText = `Disconnected (${reason})`;
                btnConnect.innerText = 'Connect';
                btnConnect.className = '';
                btnConnect.disabled = false;
                addLog('SYSTEM', { status: 'Disconnected', reason });
            });

            socket.on('connect_error', (err) => {
                statusDot.className = 'status-dot error';
                statusText.innerText = `Connection Error: ${err.message}`;
                btnConnect.innerText = 'Connect';
                btnConnect.className = '';
                btnConnect.disabled = false;
                addLog('ERROR', { message: err.message, context: err.context });
            });

            // Listen for implemented notification events
            socket.on('NEW_NOTIFICATION', (data) => {
                addLog('NEW_NOTIFICATION', data);
                updateBadges(data.unreadCount);
            });

            socket.on('NOTIFICATION_READ_SYNC', (data) => {
                addLog('NOTIFICATION_READ_SYNC', data);
                updateBadges(data.unreadCount);
            });

            socket.on('NOTIFICATION_READ_ALL_SYNC', (data) => {
                addLog('NOTIFICATION_READ_ALL_SYNC', data);
                updateBadges(data.unreadCount);
            });
        };

        document.getElementById('btnClear').onclick = () => {
            logContent.innerHTML = '<div class="log-entry" style="color: #64748b; text-align: center; padding: 40px;">Log cleared.</div>';
        };
    </script>
